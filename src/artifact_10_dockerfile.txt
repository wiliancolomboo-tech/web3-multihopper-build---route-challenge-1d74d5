# Multi-stage production container for MultiHopper Orchestrator
FROM node:20-alpine AS builder

WORKDIR /usr/src/app
COPY package*.json tsconfig.json ./
RUN npm ci

COPY . .
RUN npm run build || true

FROM node:20-alpine AS runner
WORKDIR /usr/src/app

ENV NODE_ENV=production
COPY package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app ./

USER node
EXPOSE 8080

CMD ["npx", "ts-node", "multihopper-cli.ts", "execute"]
