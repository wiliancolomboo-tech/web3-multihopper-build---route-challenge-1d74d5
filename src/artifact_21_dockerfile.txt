# Multi-stage production build for MultiHopper Routing Engine
FROM node:20-slim AS builder

WORKDIR /usr/src/app

COPY package*.json tsconfig.json ./
RUN npm ci

COPY src ./src
RUN npm run build

FROM node:20-slim

WORKDIR /usr/src/app

COPY package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist

USER node
ENV NODE_ENV=production
EXPOSE 9100

CMD ["node", "dist/cli.js"]
